# srvctl Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `srvctl`, a single static Go binary that reads a YAML job file and a `servers.yaml` list of SSH config aliases, runs tasks over SSH, and reports per-server results.

**Architecture:**
- CLI parsing via `spf13/cobra` with a `run` subcommand.
- A loader package parses `servers.yaml` and `jobs.yaml`.
- An SSH config resolver maps server aliases to connection details using `github.com/kevinburke/ssh_config`.
- A runner package builds per-target sessions, uploads scripts via SFTP, executes commands over SSH, and collects results.
- A reporter package formats output as text tables or JSON.

**Tech Stack:** Go 1.22+, `golang.org/x/crypto/ssh`, `golang.org/x/crypto/ssh/knownhosts`, `github.com/kevinburke/ssh_config`, `gopkg.in/yaml.v3`, `github.com/fatih/color`, `github.com/spf13/cobra`.

Create `internal/config/load.go`:

```go
package config

import (
    "fmt"
    "os"

    "gopkg.in/yaml.v3"
)

func LoadJob(path string) (*Job, error) {
    b, err := os.ReadFile(path)
    if err != nil {
        return nil, err
    }
    var j Job
    if err := yaml.Unmarshal(b, &j); err != nil {
        return nil, err
    }
    if err := j.Validate(); err != nil {
        return nil, err
    }
    return &j, nil
}

func LoadServers(path string) (*Servers, error) {
    b, err := os.ReadFile(path)
    if err != nil {
        return nil, err
    }
    var s Servers
    if err := yaml.Unmarshal(b, &s); err != nil {
        return nil, err
    }
    return &s, nil
}

func (j *Job) Validate() error {
    if j.Name == "" {
        return fmt.Errorf("job name is required")
    }
    if len(j.Tasks) == 0 {
        return fmt.Errorf("job must have at least one task")
    }
    for i, t := range j.Tasks {
        if t.Name == "" {
            return fmt.Errorf("task %d: name is required", i)
        }
        if t.Command == "" && t.Script == "" && t.Shell == "" {
            return fmt.Errorf("task %s: one of command, script, or shell is required", t.Name)
        }
    }
    return nil
}

// UnmarshalYAML allows `servers: ./servers.yaml` or `servers: [a, b]`.
func (s *ServersSource) UnmarshalYAML(node *yaml.Node) error {
    s.IsSet = true
    switch node.Kind {
    case yaml.ScalarNode:
        s.Path = node.Value
    case yaml.SequenceNode:
        s.List = make([]string, len(node.Content))
        for i, n := range node.Content {
            s.List[i] = n.Value
        }
    default:
        return fmt.Errorf("servers must be a file path or a list of aliases")
    }
    return nil
}
```

```go
package config

import (
    "fmt"
    "os"
    "path/filepath"
    "strconv"

    ssh_config "github.com/kevinburke/ssh_config"
)

// ResolveServers reads ssh_config and returns Server structs for aliases.
func ResolveServers(aliases []string, configPath string) ([]Server, error) {
    if configPath == "" {
        home, err := os.UserHomeDir()
        if err != nil {
            return nil, err
        }
        configPath = filepath.Join(home, ".ssh", "config")
    }

    cfg, err := ssh_config.Decode(configPath)
    if err != nil {
        return nil, fmt.Errorf("parse ssh config: %w", err)
    }

    var out []Server
    for _, alias := range aliases {
        user, _ := cfg.Get(alias, "User")
        if user == "" {
            user = os.Getenv("USER")
        }
        hostname, _ := cfg.Get(alias, "HostName")
        if hostname == "" {
            hostname = alias
        }
        portStr, _ := cfg.Get(alias, "Port")
        if portStr == "" {
            portStr = "22"
        }
        if _, err := strconv.Atoi(portStr); err != nil {
            return nil, fmt.Errorf("server %s: invalid port %q", alias, portStr)
        }
        key, _ := cfg.Get(alias, "IdentityFile")

        out = append(out, Server{
            Alias:        alias,
            HostName:     hostname,
            User:         user,
            Port:         portStr,
            IdentityFile: key,
        })
    }
    return out, nil
}
```

```go
package template

import (
    "bytes"

    "text/template"
)

func Render(s string, vars map[string]string) (string, error) {
    if len(vars) == 0 {
        return s, nil
    }
    t, err := template.New("task").Option("missingkey=error").Parse(s)
    if err != nil {
        return "", fmt.Errorf("template parse: %w", err)
    }
    var buf bytes.Buffer
    if err := t.Execute(&buf, vars); err != nil {
        return "", fmt.Errorf("template execute: %w", err)
    }
    return buf.String(), nil
}
```

```go
package ssh

import (
    "fmt"
    "net"
    "os"
    "path/filepath"

    "golang.org/x/crypto/ssh"
    "golang.org/x/crypto/ssh/agent"
    "golang.org/x/crypto/ssh/knownhosts"
)

type AuthOptions struct {
    User            string
    IdentityFile    string
    Password        string
    IgnoreHostKey   bool
    KnownHostsPath  string
}

func buildAuthMethods(opts AuthOptions) ([]ssh.AuthMethod, error) {
    var methods []ssh.AuthMethod

    if opts.Password != "" {
        methods = append(methods, ssh.Password(opts.Password))
    }

    if opts.IdentityFile != "" {
        keyPath := opts.IdentityFile
        if keyPath[:2] == "~/" {
            home, err := os.UserHomeDir()
            if err != nil {
                return nil, err
            }
            keyPath = filepath.Join(home, keyPath[2:])
        }
        key, err := os.ReadFile(keyPath)
        if err != nil {
            return nil, fmt.Errorf("read identity file: %w", err)
        }
        signer, err := ssh.ParsePrivateKey(key)
        if err != nil {
            return nil, fmt.Errorf("parse private key: %w", err)
        }
        methods = append(methods, ssh.PublicKeys(signer))
    }

    if sock := os.Getenv("SSH_AUTH_SOCK"); sock != "" {
        conn, err := net.Dial("unix", sock)
        if err == nil {
            methods = append(methods, ssh.PublicKeysCallback(agent.NewClient(conn).Signers))
        }
    }

    if len(methods) == 0 {
        return nil, fmt.Errorf("no authentication methods available")
    }
    return methods, nil
}

func hostKeyCallback(opts AuthOptions) (ssh.HostKeyCallback, error) {
    if opts.IgnoreHostKey {
        return ssh.InsecureIgnoreHostKey(), nil
    }
    path := opts.KnownHostsPath
    if path == "" {
        home, err := os.UserHomeDir()
        if err != nil {
            return nil, err
        }
        path = filepath.Join(home, ".ssh", "known_hosts")
    }
    cb, err := knownhosts.New(path)
    if err != nil {
        return nil, fmt.Errorf("known_hosts: %w", err)
    }
    return cb, nil
}
```

```go
package ssh

import (
    "fmt"
    "strconv"

    "github.com/wk/srvctl/internal/config"
    "golang.org/x/crypto/ssh"
)

func NewClient(server config.Server, opts AuthOptions) (*ssh.Client, error) {
    methods, err := buildAuthMethods(opts)
    if err != nil {
        return nil, err
    }
    hostKey, err := hostKeyCallback(opts)
    if err != nil {
        return nil, err
    }
    port, err := strconv.Atoi(server.Port)
    if err != nil {
        return nil, err
    }
    cfg := &ssh.ClientConfig{
        User:            server.User,
        Auth:            methods,
        HostKeyCallback: hostKey,
    }
    addr := fmt.Sprintf("%s:%d", server.HostName, port)
    return ssh.Dial("tcp", addr, cfg)
}
```

```go
package ssh

import (
    "bytes"
    "context"
    "fmt"
    "io"

    "golang.org/x/crypto/ssh"
)

type ExecResult struct {
    ExitCode int
    Stdout   string
    Stderr   string
    Duration string
}

func Exec(ctx context.Context, client *ssh.Client, cmd string, env map[string]string) (*ExecResult, error) {
    session, err := client.NewSession()
    if err != nil {
        return nil, fmt.Errorf("new session: %w", err)
    }
    defer session.Close()

    for k, v := range env {
        if err := session.Setenv(k, v); err != nil {
            // Some servers reject env; continue on error.
            _ = err
        }
    }

    var stdout, stderr bytes.Buffer
    session.Stdout = &stdout
    session.Stderr = &stderr

    if err := session.Run(cmd); err != nil {
        if exitErr, ok := err.(*ssh.ExitError); ok {
            return &ExecResult{
                ExitCode: exitErr.ExitStatus(),
                Stdout:   stdout.String(),
                Stderr:   stderr.String(),
            }, nil
        }
        return nil, fmt.Errorf("run: %w", err)
    }

    return &ExecResult{
        ExitCode: 0,
        Stdout:   stdout.String(),
        Stderr:   stderr.String(),
    }, nil
}
```

```go
package ssh

import (
    "crypto/rand"
    "encoding/hex"
    "fmt"
    "os"
    "path/filepath"
    "strings"

    "github.com/pkg/sftp"
    "golang.org/x/crypto/ssh"
)

func UploadScript(client *ssh.Client, interpreter string, scriptName string, content []byte) (remotePath string, cleanup func(), err error) {
    var ext string
    switch interpreter {
    case "powershell", "pwsh":
        ext = ".ps1"
    default:
        ext = ".sh"
    }

    b := make([]byte, 8)
    _, _ = rand.Read(b)
    hash := hex.EncodeToString(b)

    var remote string
    if strings.HasPrefix(interpreter, "powershell") || interpreter == "pwsh" {
        remote = fmt.Sprintf(`C:\Windows\Temp\srvctl-%s-%s%s`, hash, scriptName, ext)
    } else {
        remote = fmt.Sprintf("/tmp/srvctl-%s-%s%s", hash, scriptName, ext)
    }

    cleanup = func() { _ = Exec(context.Background(), client, fmt.Sprintf("rm %s", remote), nil) }

    sftpClient, err := sftp.NewClient(client)
    if err != nil {
        return "", nil, fmt.Errorf("sftp client: %w", err)
    }
    defer sftpClient.Close()

    f, err := sftpClient.Create(remote)
    if err != nil {
        return "", nil, fmt.Errorf("sftp create: %w", err)
    }
    if _, err := f.Write(content); err != nil {
        f.Close()
        return "", nil, fmt.Errorf("sftp write: %w", err)
    }
    if err := f.Close(); err != nil {
        return "", nil, fmt.Errorf("sftp close: %w", err)
    }

    return remote, cleanup, nil
}
```

```go
    TaskResults []TaskResult
}

func (r *JobResult) HighestExitCode() int {
    max := 0
    for _, tr := range r.TaskResults {
        if tr.ExitCode > max {
            max = tr.ExitCode
        }
    }
    return max
}
```

```go
package runner

import (
    "context"
    "fmt"
    "os"
    "sync"
    "time"

    "github.com/wk/srvctl/internal/config"
    "github.com/wk/srvctl/internal/ssh"
    "github.com/wk/srvctl/internal/template"
    "golang.org/x/crypto/ssh"
)

type Options struct {
    Job          *config.Job
    Servers      []config.Server
    Workers      int
    FailFast     bool
    Keep         bool
    IgnoreHostKey bool
    SSHConfig    string
}

func Run(ctx context.Context, opts Options) (*JobResult, error) {
    if opts.Workers <= 0 {
        opts.Workers = 10
    }

    clients := make(map[string]*ssh.Client)
    var mu sync.Mutex
    var err error

    // Open all clients up front so failures are detected early.
    for _, s := range opts.Servers {
        c, e := ssh.NewClient(s, ssh.AuthOptions{
            User:          s.User,
            IdentityFile:  s.IdentityFile,
            IgnoreHostKey: opts.IgnoreHostKey,
        })
        if e != nil {
            // store nil, report later per task
            mu.Lock()
            clients[s.Alias] = nil
            mu.Unlock()
        } else {
            mu.Lock()
            clients[s.Alias] = c
            mu.Unlock()
        }
    }

    var results []TaskResult
    var resMu sync.Mutex

    for _, task := range opts.Job.Tasks {
        taskServers := opts.Servers
        if len(task.Servers) > 0 {
            taskServers = filterServers(opts.Servers, task.Servers)
        }

        var wg sync.WaitGroup
        sem := make(chan struct{}, opts.Workers)
        cancel := make(chan struct{})
        var once sync.Once

        for _, server := range taskServers {
            wg.Add(1)
            sem <- struct{}{}
            go func(srv config.Server, t config.Task) {
                defer wg.Done()
                defer func() { <-sem }()

                select {
                case <-cancel:
                    resMu.Lock()
                    results = append(results, TaskResult{
                        TaskName: t.Name,
                        Server:   srv.Alias,
                        Error:    "skipped due to fail-fast",
                    })
                    resMu.Unlock()
                    return
                default:
                }

                client := clients[srv.Alias]
                if client == nil {
                    resMu.Lock()
                    results = append(results, TaskResult{
                        TaskName: t.Name,
                        Server:   srv.Alias,
                        Error:    "connection failed",
                    })
                    resMu.Unlock()
                    if opts.FailFast {
                        once.Do(func() { close(cancel) })
                    }
                    return
                }

                start := time.Now()
                exit, stdout, stderr, runErr := runTask(ctx, client, opts.Job, srv, t, opts.Keep)
                dur := time.Since(start).Milliseconds()

                resMu.Lock()
                results = append(results, TaskResult{
                    TaskName: t.Name,
                    Server:   srv.Alias,
                    ExitCode: exit,
                    Stdout:   stdout,
                    Stderr:   stderr,
                    Error:    errString(runErr),
                    Duration: dur,
                })
                resMu.Unlock()

                if opts.FailFast && (exit != 0 || runErr != nil) {
                    once.Do(func() { close(cancel) })
                }
            }(server, task)
        }
        wg.Wait()

        // Check if we should stop.
        if opts.FailFast {
            for _, r := range results[len(results)-len(taskServers):] {
                if r.ExitCode != 0 || r.Error != "" {
                    return &JobResult{TaskResults: results}, fmt.Errorf("fail-fast: task %s failed on %s", r.TaskName, r.Server)
                }
            }
        }
    }

    return &JobResult{TaskResults: results}, nil
}

func runTask(ctx context.Context, client *ssh.Client, job *config.Job, server config.Server, task config.Task, keep bool) (int, string, string, error) {
    interpreter := task.Interpreter
    if interpreter == "" {
        interpreter = job.Defaults.Interpreter
    }

    env := mergeEnv(job.Env, task.Env)
    vars := job.Vars

    switch {
    case task.Command != "":
        rendered, err := template.Render(task.Command, vars)
        if err != nil {
            return 0, "", "", err
        }
        cmd := fmt.Sprintf("%s -c %q", interpreter, rendered)
        if interpreter == "powershell" || interpreter == "pwsh" || interpreter == "cmd" {
            cmd = fmt.Sprintf("%s -Command %q", interpreter, rendered)
        }
        res, err := ssh.Exec(ctx, client, cmd, env)
        if err != nil {
            return 0, "", "", err
        }
        return res.ExitCode, res.Stdout, res.Stderr, nil

    case task.Shell != "":
        rendered, err := template.Render(task.Shell, vars)
        if err != nil {
            return 0, "", "", err
        }
        remote, cleanup, err := ssh.UploadScript(client, interpreter, task.Name, []byte(rendered))
        if err != nil {
            return 0, "", "", err
        }
        if !keep {
            defer cleanup()
        }
        cmd := fmt.Sprintf("%s %s", interpreter, remote)
        if interpreter == "powershell" || interpreter == "pwsh" {
            cmd = fmt.Sprintf("%s -ExecutionPolicy Bypass -File %s", interpreter, remote)
        }
        res, err := ssh.Exec(ctx, client, cmd, env)
        if err != nil {
            return 0, "", "", err
        }
        return res.ExitCode, res.Stdout, res.Stderr, nil

    case task.Script != "":
        b, err := os.ReadFile(task.Script)
        if err != nil {
            return 0, "", "", err
        }
        remote, cleanup, err := ssh.UploadScript(client, interpreter, task.Name, b)
        if err != nil {
            return 0, "", "", err
        }
        if !keep {
            defer cleanup()
        }
        cmd := fmt.Sprintf("bash %s", remote)
        if interpreter == "powershell" || interpreter == "pwsh" {
            cmd = fmt.Sprintf("%s -ExecutionPolicy Bypass -File %s", interpreter, remote)
        } else if interpreter != "" {
            cmd = fmt.Sprintf("%s %s", interpreter, remote)
        } else {
            cmd = fmt.Sprintf("bash %s", remote)
        }
        res, err := ssh.Exec(ctx, client, cmd, env)
        if err != nil {
            return 0, "", "", err
        }
        return res.ExitCode, res.Stdout, res.Stderr, nil
    }

    return 0, "", "", fmt.Errorf("no runnable content in task %s", task.Name)
}

func mergeEnv(global, task map[string]string) map[string]string {
    out := make(map[string]string)
    for k, v := range global {
        out[k] = v
    }
    for k, v := range task {
        out[k] = v
    }
    return out
}

func filterServers(servers []config.Server, names []string) []config.Server {
    allowed := make(map[string]struct{}, len(names))
    for _, n := range names {
        allowed[n] = struct{}{}
    }
    var out []config.Server
    for _, s := range servers {
        if _, ok := allowed[s.Alias]; ok {
            out = append(out, s)
        }
    }
    return out
}

func errString(err error) string {
    if err == nil {
        return ""
    }
    return err.Error()
}
```
